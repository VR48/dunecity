cmake_minimum_required(VERSION 3.20)
project(DuneCityDirectSessionSmoke LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(SDL2 CONFIG REQUIRED)
find_package(CURL REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(LibDataChannel CONFIG REQUIRED)
get_filename_component(DUNE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(direct_backend OBJECT "${DUNE_ROOT}/src/Network/DirectPeerConnectionLibdatachannel.cpp")
target_compile_definitions(direct_backend PUBLIC DUNECITY_HAVE_LIBDATACHANNEL=1)
target_include_directories(direct_backend PUBLIC "${DUNE_ROOT}/include")
target_link_libraries(direct_backend PUBLIC SDL2::SDL2 LibDataChannel::LibDataChannel)
add_executable(session-peer peer.cpp "${DUNE_ROOT}/src/Network/DirectRoomTransport.cpp"
               "${DUNE_ROOT}/src/Network/BoundedHttpClient.cpp")
target_link_libraries(session-peer PRIVATE direct_backend CURL::libcurl)
add_executable(channel-policy channel-policy.cpp)
target_link_libraries(channel-policy PRIVATE direct_backend)
enable_testing()
add_test(NAME channel-policy COMMAND channel-policy)
